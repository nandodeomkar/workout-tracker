export function buildWarmupSets(workingWeight: number, barWeight: number, unit: 'kg' | 'lb'): WarmupStep[] {

### Sparkline (new SVG component, no analog)

**No analog exists in this repo.** `react-native-svg` is not used anywhere today (RESEARCH.md A3/Pitfall 1 — confirmed absent even transitively; victory-native is Skia-based, not SVG). Build this as a small net-new leaf component (e.g. `components/Sparkline.tsx`) using the pattern RESEARCH.md Part A3 already worked out in full (Polyline + baseline Line, flat-line no-history branch) — copy that code block directly rather than re-deriving it:
```tsx
import Svg, { Polyline, Line } from 'react-native-svg';

function Sparkline({ points, color }: { points: number[]; color: string }) {
  const W = 76, H = 26;
  if (points.length === 0) {
    return (
      <Svg width={W} height={H}>
        <Line x1={0} y1={H / 2} x2={W} y2={H / 2} stroke="#7d7979" strokeWidth={1.6} />
      </Svg>
    );
  }
  const min = Math.min(...points);
  const max = Math.max(...points);
  const range = max - min || 1;
  const step = points.length > 1 ? W / (points.length - 1) : 0;
  const coords = points
    .map((v, i) => `${i * step},${H - ((v - min) / range) * H}`)
    .join(' ');
  return (
    <Svg width={W} height={H}>
      <Line x1={0} y1={H} x2={W} y2={H} stroke="rgba(245,242,239,0.14)" strokeWidth={1} />
      <Polyline points={coords} fill="none" stroke={color} strokeWidth={1.6} />
    </Svg>
  );
}
```
Data comes from existing `lib/chartSeries.ts`'s `buildChartSeries(await loadExerciseChartSets(db, exerciseId), 'e1rm')` — `.slice(-8).map(p => p.value)` per UI-SPEC OQ-3.

**Colocated test expected:** NO for the component; the delta-% calculation feeding it (last-vs-previous session e1RM) is simple enough to inline, but if extracted to a named function it should get a colocated test mirroring `lib/chartSeries.test.ts`'s existing `describe('buildChartSeries', ...)` structure.
